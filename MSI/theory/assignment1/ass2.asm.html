; ass2.asm 

.MODEL SMALL 
.STACK 100h 
.DATA 
    dummy        DB 275 DUP(0) 
    arr          DB '9','1','0','2','8','5','4','7','3','6'      ; array of 10 elements 
    msg_unsorted DB 0Ah,0Dh,'Unsorted array: $' 
    msg_sorted   DB 0Ah,0Dh,'Sorted array: $' 
    newline      DB 0Ah,0Dh,'$' 
    space        DB ' $' 
 
.CODE 
MAIN PROC 
    ; --- initialize data segment ---  
    MOV AX, @DATA 
    MOV DS, AX               
    ; --- display unsorted array --- 
    LEA DX, msg_unsorted 
    MOV AH, 09h 
    INT 21h 
    CALL PRINT_ARRAY 
 
    ; --- sort array using bubble sort --- 
    CALL BUBBLE_SORT 
 
    ; --- display sorted array --- 
    LEA DX, msg_sorted 
    MOV AH, 09h 
    INT 21h 
    CALL PRINT_ARRAY 
 
    ; --- exit program --- 
    MOV AH, 4Ch 
    INT 21h 
MAIN ENDP 
 
 
; SUBROUTINES 
 
 
; --- Print Array Subroutine --- 
PRINT_ARRAY PROC 
    MOV SI, 0 
    MOV CX, 10               ; total 10 elements 
PRINT_LOOP: 
    MOV AL, arr[SI] 
    ;ADD AL, 30h              ; convert to ASCII 
    MOV DL, AL 
    MOV AH, 02h 
    INT 21h 
 
    ; print space for readability 
    LEA DX, space 
    MOV AH, 09h 
    INT 21h 
 
    INC SI 
    LOOP PRINT_LOOP 
 
    ; newline 
    LEA DX, newline 
    MOV AH, 09h 
    INT 21h 
    RET 
PRINT_ARRAY ENDP 
 
 
; --- Swap Subroutine --- 
; expects SI to point to first element of pair 
SWAP PROC 
    MOV AL, arr[SI]          ; arr[SI] -> AL 
    MOV BL, arr[SI+1]        ; arr[SI+1] -> BL 
 
    MOV arr[SI], BL 
    MOV arr[SI+1], AL 
    RET 
SWAP ENDP 
 
 
; --- Bubble Sort Subroutine --- 
BUBBLE_SORT PROC 
    MOV CX, 9                ; outer loop count (n-1) 
OUTER_LOOP: 
    MOV SI, 0 
    MOV DX, CX               ; inner loop count 
 
INNER_LOOP: 
    MOV AL, arr[SI] 
    MOV BL, arr[SI+1] 
    CMP AL, BL 
    JBE SKIP_SWAP            ; skip if already in order 
    CALL SWAP                ; otherwise, swap elements 
 
SKIP_SWAP: 
    INC SI 
    DEC DX 
    JNZ INNER_LOOP     ; complete the inner loop count first jump not zero 
    LOOP OUTER_LOOP 
    RET 
BUBBLE_SORT ENDP 
 
END MAIN
